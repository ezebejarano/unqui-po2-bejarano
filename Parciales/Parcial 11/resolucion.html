Enunciado (resumen, reconstruido)

El Ministerio de Planificación necesita el monto total a invertir por las secretarías que dependen de él. Toda secretaría entiende montoTotal().

Una de ellas gestiona obras. Toda obra tiene nombre, fecha de inicio y un empleado líder. Hay dos tipos:

  • Proyecto (obra elemental): duración en días + un conjunto de empleados (nombre, rol, sueldo diario). Inversión = suma de sueldos diarios × duración, más el líder, que cobra un 30% extra.
  • Programa: agrupa obras (proyectos u otros programas). Inversión = suma de sus obras + un costo fijo de gestión. Su fecha de inicio es la más temprana de las de sus obras.

Además ya existe la clase SecretariaDeInfraestructura, desarrollada por otro equipo, que sabe responder inversionTotal() y no se puede modificar. El Ministerio necesita pedirle su monto igual que a las demás.

Tareas: 1) UML · 2) patrones y roles · 3) código Java · 4) script de instanciación.

Actividad 2 — Patrones y roles

ClaseRol
ObraComponent. Es clase abstracta (no interfaz) porque tiene estado (nombre, fecha, líder) y constructor.

Ojo con la pregunta teórica que se cruza acá: "¿siempre puedo reemplazar una clase abstracta raíz por una interfaz?" → FALSO, y Obra es el ejemplo: tiene atributos de instancia y constructor, y una interfaz no puede tener ninguna de las dos cosas.
El Composite no sólo suma plata: también sobrescribe getFechaDeInicio() devolviendo el mínimo de sus hijos. Un compuesto puede redefinir cualquier consulta, no sólo el total.

Actividad 3 — Código Java

import java.time.LocalDate;
import java.util.*;

interface Empleado {
    String nombre();
    String rol();
    double sueldoDiario();
}

class EmpleadoPublico implements Empleado {
    private String nombre, rol; private double sueldoDiario;
    public EmpleadoPublico(String nombre, String rol, double sueldoDiario) {
        this.nombre=nombre; this.rol=rol; this.sueldoDiario=sueldoDiario;
    }
    public String nombre()       { return nombre; }
    public String rol()          { return rol; }
    public double sueldoDiario() { return sueldoDiario; }
}

// ================= COMPOSITE (obras) =================
abstract class Obra {                                // Rol: COMPONENT (abstracta: tiene ESTADO)
    private String nombre;
    private LocalDate fechaDeInicio;
    private Empleado lider;

    public Obra(String nombre, LocalDate fechaDeInicio, Empleado lider) {
        this.nombre=nombre; this.fechaDeInicio=fechaDeInicio; this.lider=lider;
    }
    public String getNombre()  { return nombre; }
    public Empleado getLider() { return lider; }
    public LocalDate getFechaDeInicio() { return fechaDeInicio; }

    public abstract double inversionTotal();
}

class Proyecto extends Obra {                        // Rol: LEAF
    private int duracionEnDias;
    private List<Empleado> empleados;

    public Proyecto(String nombre, LocalDate fechaDeInicio, Empleado lider,
                    List<Empleado> empleados, int duracionEnDias) {
        super(nombre, fechaDeInicio, lider);
        this.empleados = new ArrayList<>(empleados);
        this.duracionEnDias = duracionEnDias;
    }
    public double inversionTotal() { return sueldoDeLosEmpleados() + sueldoDelLider(); }

    private double sueldoDeLosEmpleados() {
        return empleados.stream().mapToDouble(Empleado::sueldoDiario).sum() * duracionEnDias;
    }
    private double sueldoDelLider() {
        return getLider().sueldoDiario() * duracionEnDias * 1.30;      // el lider cobra 30% extra
    }
}

class Programa extends Obra {                        // Rol: COMPOSITE
    private List<Obra> obras = new ArrayList<>();
    private double costoFijoDeGestion;

    public Programa(String nombre, LocalDate fechaDeInicio, Empleado lider, double costoFijoDeGestion) {
        super(nombre, fechaDeInicio, lider);
        this.costoFijoDeGestion = costoFijoDeGestion;
    }
    public void agregarObra(Obra obra)  { obras.add(obra); }
    public void eliminarObra(Obra obra) { obras.remove(obra); }

    public double inversionTotal() {                                   // recursivo (puede haber programas)
        return costoFijoDeGestion + obras.stream().mapToDouble(Obra::inversionTotal).sum();
    }
    @Override
    public LocalDate getFechaDeInicio() {                             // la MAS TEMPRANA de sus obras
        return obras.stream().map(Obra::getFechaDeInicio)
                    .min(Comparator.naturalOrder())
                    .orElse(super.getFechaDeInicio());
    }
}

// ================= ADAPTER (secretarias) =================
interface Secretaria {                               // Rol: TARGET
    double montoTotal();
}

class SecretariaDeObrasPublicas implements Secretaria {     // ya habla el protocolo del target
    private String nombre;
    private List<Obra> obras = new ArrayList<>();
    public SecretariaDeObrasPublicas(String nombre) { this.nombre = nombre; }
    public void agregarObra(Obra obra) { obras.add(obra); }
    public double montoTotal() { return obras.stream().mapToDouble(Obra::inversionTotal).sum(); }
}

class SecretariaDeInfraestructura {                  // Rol: ADAPTEE (otro protocolo: NO se toca)
    private List<Obra> obras = new ArrayList<>();
    public void agregarObra(Obra obra) { obras.add(obra); }
    public double inversionTotal() { return obras.stream().mapToDouble(Obra::inversionTotal).sum(); }
}

class SecretariaDeInfraestructuraAdapter implements Secretaria {   // Rol: ADAPTER
    private SecretariaDeInfraestructura secretaria;
    public SecretariaDeInfraestructuraAdapter(SecretariaDeInfraestructura s) { this.secretaria = s; }
    public double montoTotal() { return secretaria.inversionTotal(); }   // traduce la llamada
}

class MinisterioDePlanificacion {                    // Rol: CLIENT (solo conoce la INTERFAZ)
    private List<Secretaria> secretarias = new ArrayList<>();
    public void agregarSecretaria(Secretaria s) { secretarias.add(s); }
    public double montoTotalAInvertir() {
        return secretarias.stream().mapToDouble(Secretaria::montoTotal).sum();
    }
}

Actividad 4 — Instanciación (cuentas verificadas)

Empleado lider   = new EmpleadoPublico("Ana",  "Arquitecta", 50000);
Empleado albanil = new EmpleadoPublico("Luis", "Albanil",    30000);
Empleado plomero = new EmpleadoPublico("Sol",  "Plomera",    35000);

// LEAF: (30.000+35.000)*10 = 650.000  +  50.000*10*1,30 = 650.000   => 1.300.000
Proyecto escuela = new Proyecto("Escuela N5", LocalDate.of(2026,3,1), lider,
                                Arrays.asList(albanil, plomero), 10);
// LEAF: 30.000*5 = 150.000  +  50.000*5*1,30 = 325.000              =>   475.000
Proyecto plaza   = new Proyecto("Plaza Central", LocalDate.of(2026,1,15), lider,
                                Arrays.asList(albanil), 5);

// COMPOSITE: 200.000 + 1.300.000 + 475.000                          => 1.975.000
Programa programaBarrial = new Programa("Programa Barrial", LocalDate.of(2026,6,1), lider, 200000);
programaBarrial.agregarObra(escuela);
programaBarrial.agregarObra(plaza);
programaBarrial.getFechaDeInicio();     // 2026-01-15  <- el MIN de sus hijos (NO su fecha propia)

// COMPOSITE dentro de COMPOSITE: 500.000 + 1.975.000                => 2.475.000
Programa programaProvincial = new Programa("Programa Provincial", LocalDate.of(2026,6,1), lider, 500000);
programaProvincial.agregarObra(programaBarrial);

SecretariaDeObrasPublicas obrasPublicas = new SecretariaDeObrasPublicas("Obras Publicas");
obrasPublicas.agregarObra(programaProvincial);

SecretariaDeInfraestructura infraestructura = new SecretariaDeInfraestructura();   // el ADAPTEE
infraestructura.agregarObra(plaza);                                                // 475.000
Secretaria infraAdaptada = new SecretariaDeInfraestructuraAdapter(infraestructura);

MinisterioDePlanificacion ministerio = new MinisterioDePlanificacion();
ministerio.agregarSecretaria(obrasPublicas);
ministerio.agregarSecretaria(infraAdaptada);        // entra como una secretaria mas

ministerio.montoTotalAInvertir();       // 2.950.000,00   (2.475.000 + 475.000)
